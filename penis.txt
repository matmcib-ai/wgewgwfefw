local __SXE_GrappleFired = false
local __SXE_KillQ = {} 

do
    local RS = game:GetService("ReplicatedStorage")
    local KEEP = {
        CustomProximityPromptController = true, InteractController = true,
        PlotController = true, AnimalController = true, PetController = true,
        WorldBrainrotController = true, EquipBrainrotsController = true,
        EquipBrainrotsRenderController = true, SelectBrainrotsController = true,
        BrainrotPreloadingController = true, AssetStreamController = true,
        AdminPanelController = true, NotificationController = true,
    }

    local function wrap(fn)
        if newcclosure then return newcclosure(fn) end
        return fn
    end

    local function isCtrlFolder(x)
        return typeof(x) == "Instance" and x.Name == "Controllers" and x.Parent == RS
    end

    local oldNc
    pcall(function()
        oldNc = hookmetamethod(game, "__namecall", wrap(function(self, ...)
            local method = getnamecallmethod()
            if type(method) ~= "string" or method:lower() ~= "getchildren" then
                return oldNc(self, ...)
            end
            if not isCtrlFolder(self) then return oldNc(self, ...) end
            local s = getcallingscript()
            if not s or tostring(s.Name) ~= "ClientLoader" then return oldNc(self, ...) end
            local kids = oldNc(self, ...)
            if type(kids) ~= "table" then return kids end
            local out = {}
            for i = 1, #kids do
                local k = kids[i]
                if k and KEEP[k.Name] then out[#out + 1] = k end
            end
            return out
        end))
    end)
    task.delay(4, function() pcall(function() hookmetamethod(game, "__namecall", oldNc) end) end)
end

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local player = Players.LocalPlayer or Players.PlayerAdded:Wait()

local function optimizeAndStripChar(char)
    if not char then return end
    
    local function cleanObject(obj)
        pcall(function()
            if obj:IsA("Accessory") or obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") or obj:IsA("CharacterMesh") then
                obj:Destroy()
            elseif obj:IsA("BasePart") then
                obj.CastShadow = false
                obj.Reflectance = 0
                obj.Material = Enum.Material.SmoothPlastic
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
                obj.Enabled = false
            elseif obj:IsA("PointLight") or obj:IsA("SpotLight") then
                obj.Enabled = false
            end
        end)
    end

    task.spawn(function()
        for _, obj in ipairs(char:GetDescendants()) do
            cleanObject(obj)
        end
    end)

    char.DescendantAdded:Connect(function(obj)
        task.defer(function() cleanObject(obj) end)
    end)
end

for _, plr in ipairs(Players:GetPlayers()) do
    if plr.Character then optimizeAndStripChar(plr.Character) end
    plr.CharacterAdded:Connect(optimizeAndStripChar)
end
Players.PlayerAdded:Connect(function(plr)
    plr.CharacterAdded:Connect(optimizeAndStripChar)
end)
